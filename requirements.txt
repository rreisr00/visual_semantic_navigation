# Python dependencies for the visual-semantic navigation project
#
# Install with:
#   pip install -r requirements.txt
#
# Note: ROS 2 packages (rclpy, sensor_msgs, cv_bridge, nav2_simple_commander,
# tf2_ros) are installed via apt / colcon build and are NOT listed here.
#
# ABI constraint: the system cv_bridge (ROS Jazzy) is compiled against NumPy 1.x.
# Loading it with NumPy 2.x causes an ABI crash at import time.
# Constraint chain:
#   numpy<2            ← required by system cv_bridge (ROS Jazzy, compiled against NumPy 1.x)
#   opencv-python<4.11 ← opencv-python>=4.11 requires numpy>=2, incompatible with above

# NumPy – must stay on the 1.x ABI for cv_bridge compatibility
numpy>=1.24.0,<2

# Image processing – 4.11+ requires numpy>=2, so cap at 4.10.x
opencv-python>=4.7.0,<4.11

# Deep learning inference
torch>=2.0.0
transformers>=4.45.0
# Pillow is pulled in by transformers but listed explicitly for clarity
Pillow>=9.0.0

# Similarity computation
scipy>=1.10.0

matplotlib>=3.10.8

sentencepiece>=0.2.1

scikit-learn>=1.8.0

protobuf>=7.34.1

chromadb>=0.6.0

ultralytics>=8.0.0
